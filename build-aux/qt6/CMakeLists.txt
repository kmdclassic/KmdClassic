cmake_minimum_required(VERSION 3.22)
project(KmdClassicQtProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SKIP_RPATH ON)
if(MINGW)
    # Static Qt built with stack protection needs GCC's support runtime.
    # Keep it after the Qt archives, including in the exported link fragments.
    string(APPEND CMAKE_CXX_STANDARD_LIBRARIES " -lssp")
endif()
# DBus code generators are not used by this application. The tools we do
# execute are required explicitly below, from the native installation.
set(QT_ALLOW_MISSING_TOOLS_PACKAGES TRUE)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Widgets Network)
find_package(Qt6 QUIET COMPONENTS Test)
if(NOT USE_DBUS STREQUAL "no")
    find_package(Qt6 QUIET COMPONENTS DBus)
endif()
# Ask CMake to resolve the complete static dependency graph, including its
# installed plugin and resource initialization objects.
set(QT_NO_CREATE_VERSIONLESS_FUNCTIONS ON)
add_executable(qt_probe probe.cpp)
set_target_properties(qt_probe PROPERTIES QT_DEFAULT_PLUGINS 0)
target_link_libraries(qt_probe PRIVATE Qt6::Widgets Qt6::Network)
get_target_property(qt_type Qt6::Core TYPE)
if(qt_type STREQUAL "STATIC_LIBRARY")
    if(WIN32)
        target_link_libraries(qt_probe PRIVATE Qt6::QWindowsIntegrationPlugin)
        target_compile_definitions(qt_probe PRIVATE QT_QPA_PLATFORM_WINDOWS QT_STATICPLUGIN)
        if(TARGET Qt6::QModernWindowsStylePlugin)
            target_link_libraries(qt_probe PRIVATE Qt6::QModernWindowsStylePlugin)
            target_compile_definitions(qt_probe PRIVATE QT_WINDOWS_STYLE_MODERN)
        elseif(TARGET Qt6::QWindowsVistaStylePlugin)
            target_link_libraries(qt_probe PRIVATE Qt6::QWindowsVistaStylePlugin)
            target_compile_definitions(qt_probe PRIVATE QT_WINDOWS_STYLE_VISTA)
        endif()
    elseif(APPLE)
        target_link_libraries(qt_probe PRIVATE Qt6::QCocoaIntegrationPlugin)
        target_compile_definitions(qt_probe PRIVATE QT_QPA_PLATFORM_COCOA QT_STATICPLUGIN)
        if(TARGET Qt6::QMacStylePlugin)
            target_link_libraries(qt_probe PRIVATE Qt6::QMacStylePlugin)
        endif()
    elseif(UNIX)
        target_link_libraries(qt_probe PRIVATE Qt6::QXcbIntegrationPlugin)
        target_compile_definitions(qt_probe PRIVATE QT_QPA_PLATFORM_XCB QT_STATICPLUGIN)
    endif()
endif()
foreach(component Test DBus)
    if(TARGET Qt6::${component})
        add_executable(qt_${component}_probe probe.cpp)
        set_target_properties(qt_${component}_probe PROPERTIES QT_DEFAULT_PLUGINS 0)
        target_link_libraries(qt_${component}_probe PRIVATE Qt6::${component})
    endif()
endforeach()
find_package(Qt6 REQUIRED COMPONENTS CoreTools WidgetsTools LinguistTools)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/tools.txt" CONTENT
"MOC=$<TARGET_FILE:Qt6::moc>\nRCC=$<TARGET_FILE:Qt6::rcc>\nUIC=$<TARGET_FILE:Qt6::uic>\nLRELEASE=$<TARGET_FILE:Qt6::lrelease>\nLUPDATE=$<TARGET_FILE:Qt6::lupdate>\nLCONVERT=$<TARGET_FILE:Qt6::lconvert>\n")
